
Tablero {
    - Map<Posicion, Casilla> casillas
    - ALCANCE_MOVIMIENTO = 3;

    + obtenerCasilla(Posicion posicion)
    + moverUnidad(Posicion origen, Posicion destino)
    + movimientosValidosDesde(Posicion origen)
    + empujarUnidad(Posicion origen, Direccion direccion)
}

Casilla {
    - Terreno terreno
    - Unidad unidad

    + estaLibre()
    + puedeRecibir(Unidad unidad)
    + recibirUnidad(Unidad unidad)
    + trasladarUnidadA(Casilla destino)
    + empujarUnidadHacia(Casilla destino)
    + recibirUnidadDesplazada(Unidad unidad, Casilla origen)
    + recibirAtaque(int danio)
    + cambiarTerreno(Terreno nuevoTerreno)
}

Posicion {
    - int fila
    - int columna

    + obtenerAdyacente(Direccion direccion)
    + distanciaACelda(Posicion otra)
    - distanciaACoordenadas(int fila, int columna)
    + esIgualA(Posicion otra)
    - compararme(int fila, int columna)
    + equals(Object otro)
    + hashCode()
}

Direccion {
    ARRIBA
    ABAJO
    IZQUIERDA
    DERECHA
}

Terreno {
    + permiteIngreso(Unidad unidad)
    + recibirAtaque(int danio)
    + aplicarEfecto(Unidad unidad)
    + recibirUnidadDesplazada(Unidad unidad, Casilla destino)
}

Suelo     --|> Terreno
Agua      --|> Terreno
Acido     --|> Terreno
Montaña   --|> Terreno
Edificio  --|> Terreno

Edificio {
    - int vida
    - RedElectrica redElectrica

    + recibirAtaque(int danio)
    + recibirUnidadDesplazada(Unidad unidad, Casilla destino)
}

RedElectrica {
    - int puntos

    + restarVida(int cantidad)
    + estaAgotada()
}

Unidad {
    - int vida
    - EstadoSalud estadoSalud


    + puedeMoverse()
    + puedeAtravesar(Terreno terreno)
    + puedeTerminarEn(Terreno terreno)
    + recibirDanio(int cantidad)
    + recibirContaminacion()
    + puedeAtacar()
    + reiniciarAcciones()
}

Mech --|> Unidad
Vek  --|> Unidad

Mech {
    - Arma arma
    - boolean movimientoRealizado
    - boolean ataqueRealizado

    + atacar(Posicion origen, Posicion objetivo, Tablero tablero)
    + asentarAtaque()
    + reiniciarAcciones()
}

MechCombate    --|> Mech
MechCañon      --|> Mech
MechArtilleria --|> Mech

Vek {
    - Direccion direccionAtaqueAnunciado

    + anunciarAtaque(Direccion direccion)
    + ejecutarAtaque(Posicion origen, Tablero tablero)
}

Saltador  --|> Vek
Luciernaga --|> Vek
Avispon   --|> Vek

Arma {
    - int danio

    + atacar(Tablero tablero, Posicion origen,Posicion objetivo)
}

Puño       --|> Arma
Cañon      --|> Arma
Artilleria --|> Arma

EstadoSalud {
    SANO
    CONTAMINADO
}

Partida {
    - Tablero tablero
    - RedElectrica redElectrica
    - int turnoActual
    - Fase faseActual

    + iniciar()
    + avanzarFase()
    + avanzarTurno()
    + verificarFinDePartida()
}

Fase {
    VEK
    JUGADOR
    RESOLUCION
    EMERGENCIA
}
